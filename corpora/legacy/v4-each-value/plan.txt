(
  :"prices by fruit" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'GET'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/prices'
        )
      ),
      :query (
        %legacy:only-entries (
          $.request.query
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          200
        )
      ),
      :headers (
        :"$.Content-Type" (
          %legacy:header-value (
            $.response.headers.content-type,
            'application/json'
          )
        )
      ),
      :body (
        :"$.prices" (
          %expect:object (
            $.response.body.prices
          ),
          %flow:for-each (
            ** (
              $.response.body.prices
            ),
            %flow:if (
              %check:equals (
                ~>.key,
                'apple'
              ),
              :"$.prices.apple" (
                %match:type (
                  ~>.value,
                  1.5
                )
              ),
              :"$.prices{*}" (
                %match:type (
                  ~>.value,
                  1.5
                )
              )
            )
          )
        )
      )
    )
  )
)
