(
  :"get a greeting" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method => 'GET'
          ) => 'GET',
          'GET'
        ) => BOOL(true)
      ) => BOOL(true),
      :path (
        %match:equality (
          $.request.path => '/greeting',
          '/greeting'
        ) => BOOL(true)
      ) => BOOL(true),
      :query (
        %legacy:only-entries (
          $.request.query => {}
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true),
    :response (
      :status (
        %match:equality (
          $.response.status => 200,
          200
        ) => BOOL(true)
      ) => BOOL(true),
      :body (
        %legacy:body-type (
          $.response.headers.content-type => NULL,
          'text/plain'
        ) => BOOL(true),
        %match:regex (
          $.response.body => 'Goodbye Ada!',
          '^(?:Hello .+!)$'
        ) => ERROR(Expected 'Goodbye Ada!' to match '^(?:Hello .+!)$')
      ) => BOOL(false)
    ) => BOOL(false)
  ) => BOOL(false)
)
