(
  :"get /orders" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'GET'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/orders'
        )
      ),
      :query (
        :"$.page" (
          %expect:count (
            $.request.query.page,
            1
          ),
          %legacy:parses-as (
            $.request.query.page[0],
            'integer'
          )
        ),
        :"$.size" (
          %expect:count (
            $.request.query.size,
            1
          ),
          %legacy:parses-as (
            $.request.query.size[0],
            'integer'
          )
        ),
        %legacy:only-entries (
          $.request.query,
          'page',
          'size'
        )
      ),
      :headers (
        :"$.X-Ratio" (
          %expect:not-empty (
            $.request.headers.x-ratio
          ),
          %flow:for-each (
            ** (
              $.request.headers.x-ratio
            ),
            :"$.X-Ratio[*]" (
              %legacy:parses-as (
                ~>,
                'number'
              )
            )
          )
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          200
        )
      )
    )
  )
)
