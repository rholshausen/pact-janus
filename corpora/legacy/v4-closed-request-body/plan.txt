(
  :"create an alligator" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'POST'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/alligators'
        )
      ),
      :query (
        %legacy:only-entries (
          $.request.query
        )
      ),
      :headers (
        :"$.Content-Type" (
          %legacy:header-value (
            $.request.headers.content-type,
            'application/json'
          )
        )
      ),
      :body (
        :"$.alligator" (
          :"$.alligator.name" (
            %match:equality (
              $.request.body.alligator.name,
              'Mary'
            )
          ),
          %legacy:only-entries (
            $.request.body.alligator,
            'name'
          )
        ),
        %legacy:only-entries (
          $.request.body,
          'alligator'
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          201
        )
      )
    )
  )
)
