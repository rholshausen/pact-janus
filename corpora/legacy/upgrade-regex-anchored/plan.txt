(
  :"get a delivery date" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'GET'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/delivery'
        )
      ),
      :query (
        %legacy:only-entries (
          $.request.query
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          200
        )
      ),
      :body (
        %legacy:body-type (
          $.response.headers.content-type,
          'application/json'
        ),
        :"$.date" (
          %match:regex (
            $.response.body.date,
            '^(?:\\d{4}-\\d{2}-\\d{2})$'
          )
        )
      )
    )
  )
)
