(
  :"post /orders" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'POST'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/orders'
        )
      ),
      :query (
        %legacy:only-entries (
          $.request.query
        )
      ),
      :headers (
        :"$.Content-Type" (
          %legacy:header-value (
            $.request.headers.content-type,
            'application/xml'
          )
        )
      ),
      :body (
        %expect:object (
          $.request.body.order
        ),
        :"$.order" (
          %match:equality (
            $.request.body.order.#namespace,
            'urn:orders'
          ),
          :"$.order.@id" (
            %legacy:parses-as (
              $.request.body.order.@id,
              'integer'
            )
          ),
          :"$.order.#trimmed" (
            %flow:if (
              %check:exists (
                $.request.body.order.#trimmed
              ),
              %match:equality (
                $.request.body.order.#trimmed,
                ''
              )
            )
          ),
          %expect:count (
            $.request.body.order.item,
            2
          ),
          :"$.order.item[0]" (
            :"$.order.item[0].#trimmed" (
              %match:equality (
                $.request.body.order.item[0].#trimmed,
                'apple'
              )
            ),
            %legacy:only-entries (
              $.request.body.order.item[0],
              '#text',
              '#trimmed',
              '#namespace'
            )
          ),
          :"$.order.item[1]" (
            :"$.order.item[1].#trimmed" (
              %match:equality (
                $.request.body.order.item[1].#trimmed,
                'pear'
              )
            ),
            %legacy:only-entries (
              $.request.body.order.item[1],
              '#text',
              '#trimmed',
              '#namespace'
            )
          ),
          %legacy:only-entries (
            $.request.body.order,
            '@id',
            'item',
            '#text',
            '#trimmed',
            '#namespace'
          )
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          201
        )
      )
    )
  )
)
