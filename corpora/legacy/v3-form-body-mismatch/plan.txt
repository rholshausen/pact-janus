(
  :"get /form" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'GET'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/form'
        )
      ),
      :query (
        %legacy:only-entries (
          $.request.query
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          200
        )
      ),
      :headers (
        :"$.Content-Type" (
          %legacy:header-value (
            $.response.headers.content-type,
            'application/x-www-form-urlencoded'
          )
        )
      ),
      :body (
        :"$.tag" (
          %expect:count (
            $.response.body.tag,
            2
          ),
          %match:equality (
            $.response.body.tag[0],
            'a'
          ),
          %match:equality (
            $.response.body.tag[1],
            'b'
          )
        ),
        :"$.user_id" (
          %expect:count (
            $.response.body.user_id,
            1
          ),
          %match:regex (
            $.response.body.user_id[0],
            '^(?:[0-9]+)$'
          )
        ),
        %legacy:only-entries (
          $.response.body,
          'tag',
          'user_id'
        )
      )
    )
  )
)
