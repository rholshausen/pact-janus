(
  :"get /thing" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method => 'GET'
          ) => 'GET',
          'GET'
        ) => BOOL(true)
      ) => BOOL(true),
      :path (
        %match:equality (
          $.request.path => '/thing',
          '/thing'
        ) => BOOL(true)
      ) => BOOL(true),
      :query (
        %legacy:only-entries (
          $.request.query => {}
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true),
    :response (
      :status (
        %match:equality (
          $.response.status => 200,
          200
        ) => BOOL(true)
      ) => BOOL(true),
      :headers (
        :"$.Content-Type" (
          %legacy:header-value (
            $.response.headers.content-type => 'application/json',
            'application/json'
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true),
      :body (
        :"$.id" (
          %flow:or (
            %match:regex (
              $.response.body.id => 'abc',
              '^(?:x+)$'
            ) => ERROR(Expected 'abc' to match '^(?:x+)$'),
            %match:type (
              $.response.body.id => 'abc',
              'xxx'
            ) => BOOL(true)
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true)
  ) => BOOL(true)
)
