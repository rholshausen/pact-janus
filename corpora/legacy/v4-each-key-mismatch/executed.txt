(
  :"prices by code, keys and values both ruled" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method => 'GET'
          ) => 'GET',
          'GET'
        ) => BOOL(true)
      ) => BOOL(true),
      :path (
        %match:equality (
          $.request.path => '/prices',
          '/prices'
        ) => BOOL(true)
      ) => BOOL(true),
      :query (
        %legacy:only-entries (
          $.request.query => {}
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true),
    :response (
      :status (
        %match:equality (
          $.response.status => 200,
          200
        ) => BOOL(true)
      ) => BOOL(true),
      :headers (
        :"$.Content-Type" (
          %legacy:header-value (
            $.response.headers.content-type => 'application/json',
            'application/json'
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true),
      :body (
        :"$.prices" (
          %expect:object (
            $.response.body.prices => {'A1': 'free', 'b2': 3}
          ) => BOOL(true),
          %flow:for-each (
            ** (
              $.response.body.prices => {'A1': 'free', 'b2': 3}
            ) => [{'A1': 'free'}, {'b2': 3}],
            :"$.prices{key}" (
              %match:regex (
                ~>.key => 'A1',
                '^(?:[A-Z][0-9])$'
              ) => BOOL(true)
            ) => BOOL(true),
            :"$.prices{key}" (
              %match:regex (
                ~>.key => 'b2',
                '^(?:[A-Z][0-9])$'
              ) => ERROR(Expected 'b2' to match '^(?:[A-Z][0-9])$')
            ) => BOOL(false)
          ) => BOOL(false),
          %flow:if (
            %check:exists (
              $.response.body.prices.A1 => 'free'
            ) => BOOL(true),
            :"$.prices.A1" (
              %match:number (
                $.response.body.prices.A1 => 'free'
              ) => ERROR(Expected a number but got 'free')
            ) => BOOL(false)
          ) => BOOL(false)
        ) => BOOL(false)
      ) => BOOL(false)
    ) => BOOL(false)
  ) => BOOL(false)
)
