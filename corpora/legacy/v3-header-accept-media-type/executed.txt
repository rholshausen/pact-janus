(
  :"get one, as JSON" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method => 'GET'
          ) => 'GET',
          'GET'
        ) => BOOL(true)
      ) => BOOL(true),
      :path (
        %match:equality (
          $.request.path => '/one',
          '/one'
        ) => BOOL(true)
      ) => BOOL(true),
      :query (
        %legacy:only-entries (
          $.request.query => {}
        ) => BOOL(true)
      ) => BOOL(true),
      :headers (
        :"$.Accept" (
          %legacy:header-value (
            $.request.headers.accept => ['application/json;q=0.9'],
            'application/json'
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true),
    :response (
      :status (
        %match:equality (
          $.response.status => 200,

        ) => BOOL(true)
      ) => BOOL(true),
      :headers (
        :"$.X-Format" (
          %legacy:header-text (
            $.response.headers.x-format => ['csv; header=present'],
            'csv; header=present'
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true)
  ) => BOOL(true)
