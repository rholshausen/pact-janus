(
  :"get one, with parameters" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method => 'GET'
          ) => 'GET',
          'GET'
        ) => BOOL(true)
      ) => BOOL(true),
      :path (
        %match:equality (
          $.request.path => '/one',
          '/one'
        ) => BOOL(true)
      ) => BOOL(true),
      :query (
        %legacy:only-entries (
          $.request.query => {}
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true),
    :response (
      :status (
        %match:equality (
          $.response.status => 200,
          200
        ) => BOOL(true)
      ) => BOOL(true),
      :headers (
        :"$.X-Params" (
          %legacy:header-text (
            $.response.headers.x-params => ['a;b=1;c=2'],
            'a;b=1'
          ) => ERROR(Expected 'a;b=1;c=2' to equal 'a;b=1')
        ) => BOOL(false)
      ) => BOOL(false)
    ) => BOOL(false)
  ) => BOOL(false)
)
