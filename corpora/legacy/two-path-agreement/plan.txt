(
  :"get a count" (
    :request (
      :method (
        %match:equality (
          %upper-case (
            $.request.method
          ),
          'GET'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/count'
        )
      ),
      :query (
        %expect:only-entries (
          $.request.query
        )
      ),
      :headers ()
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          200
        )
      ),
      :headers (),
      :body (
        %match:type (
          $.response.body,
          42
        )
      )
    )
  )
)
