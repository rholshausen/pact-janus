(
  :"get a count" (
    :request (
      :method (
        %match:equality (
          %value:upper-case (
            $.request.method
          ),
          'GET'
        )
      ),
      :path (
        %match:equality (
          $.request.path,
          '/count'
        )
      ),
      :query (
        %legacy:only-entries (
          $.request.query
        )
      )
    ),
    :response (
      :status (
        %match:equality (
          $.response.status,
          200
        )
      ),
      :body (
        %legacy:body-type (
          $.response.headers.content-type,
          'application/json'
        ),
        %match:type (
          $.response.body,
          42
        )
      )
    )
  )
)
