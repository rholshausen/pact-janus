          %value:upper-case (
            $.request.method => 'GET'
          ) => 'GET',

        ) => BOOL(true)
      ) => BOOL(true),
      :path (
        %match:equality (
          $.request.path => '/note',
          '/note'
        ) => BOOL(true)
      ) => BOOL(true),
      :query (
        %legacy:only-entries (
          $.request.query => {}
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true),
    :response (
      :status (
        %match:equality (
          $.response.status => 200,
          200
        ) => BOOL(true)
      ) => BOOL(true),

          $.response.headers.content-type => ABSENT,
          'application/xml'
        ) => BOOL(true),

          ) => BOOL(false),
          :"$.note['@{urn:x}rev']" (

              '2'
            ) => BOOL(true)
          ) => BOOL(true),
          :"$.note.#text" (
            %flow:if (
              %check:exists (
                $.response.body.note.#text => ABSENT
              ) => BOOL(false),

                $.response.body.note.#text,
                ''
              )
            ) => BOOL(true)
          ) => BOOL(true),

            $.response.body.note.line => [{'#text': 'uno'}, {'#text': 'dos'}, {'#text': 'tres'}]
          ) => BOOL(true),

            ** (
              $.response.body.note.line => [{'#text': 'uno'}, {'#text': 'dos'}, {'#text': 'tres'}]
            ) => [{'#text': 'uno'}, {'#text': 'dos'}, {'#text': 'tres'}],

                ~> => {'#text': 'uno'}
              ) => BOOL(true),

                  %check:exists (
                    ~>.#text => 'uno'
                  ) => BOOL(true),
                  %match:type (
                    ~>.#text => 'uno',
                    'one'
                  ) => BOOL(true)
                ) => BOOL(true)
              ) => BOOL(true),
              %legacy:xml-only-children (
                ~> => {'#text': 'uno'}
              ) => BOOL(true)
            ) => BOOL(true),
            :"$.note.line[*]" (
              %expect:object (
                ~> => {'#text': 'dos'}
              ) => BOOL(true),
              :"$.note.line[*].#text" (

                  %check:exists (
                    ~>.#text => 'dos'
                  ) => BOOL(true),
                  %match:type (
                    ~>.#text => 'dos',
                    'one'
                  ) => BOOL(true)
                ) => BOOL(true)
              ) => BOOL(true),
              %legacy:xml-only-children (
                ~> => {'#text': 'dos'}
              ) => BOOL(true)

            :"$.note.line[*]" (

                ~> => {'#text': 'tres'}

                %flow:if (
                  %check:exists (

              ) => BOOL(true)
