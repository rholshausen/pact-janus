(
  :"submit a batch of orders" (
    :request (
      :body (
        %expect:array (
          $.request.body => [{'sku': 'SKU-1', 'qty': 1}, {'sku': 'SKU-2', 'qty': 4}, {'sku': 'SKU-3', 'qty': 2}]
        ) => BOOL(true),
        %expect:size (
          $.request.body => [{'sku': 'SKU-1', 'qty': 1}, {'sku': 'SKU-2', 'qty': 4}, {'sku': 'SKU-3', 'qty': 2}],
          2,
          NULL
        ) => BOOL(true),
        %flow:for-each (
          ** (
            $.request.body => [{'sku': 'SKU-1', 'qty': 1}, {'sku': 'SKU-2', 'qty': 4}, {'sku': 'SKU-3', 'qty': 2}]
          ) => [{'sku': 'SKU-1', 'qty': 1}, {'sku': 'SKU-2', 'qty': 4}, {'sku': 'SKU-3', 'qty': 2}],
          :"$[*]" (
            %expect:object (
              ~> => {'sku': 'SKU-1', 'qty': 1}
            ) => BOOL(true),
            :"$[*].qty" (
              %match:integer (
                ~>.qty => 1
              ) => BOOL(true)
            ) => BOOL(true),
            :"$[*].sku" (
              %match:string (
                ~>.sku => 'SKU-1'
              ) => BOOL(true)
            ) => BOOL(true)
          ) => BOOL(true),
          :"$[*]" (
            %expect:object (
              ~> => {'sku': 'SKU-2', 'qty': 4}
            ) => BOOL(true),
            :"$[*].qty" (
              %match:integer (
                ~>.qty => 4
              ) => BOOL(true)
            ) => BOOL(true),
            :"$[*].sku" (
              %match:string (
                ~>.sku => 'SKU-2'
              ) => BOOL(true)
            ) => BOOL(true)
          ) => BOOL(true),
          :"$[*]" (
            %expect:object (
              ~> => {'sku': 'SKU-3', 'qty': 2}
            ) => BOOL(true),
            :"$[*].qty" (
              %match:integer (
                ~>.qty => 2
              ) => BOOL(true)
            ) => BOOL(true),
            :"$[*].sku" (
              %match:string (
                ~>.sku => 'SKU-3'
              ) => BOOL(true)
            ) => BOOL(true)
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true)
  ) => BOOL(true)
)
