(
  :"get orders as a list" (
    :request (
      :headers (
        %expect:object (
          $.request.headers
        ),
        :"$.accept" (
          %http:list (
            $.request.headers.accept,
            {'elements': ['application/json', 'text/plain']}
          )
        ),
        :"$.vary" (
          %http:list (
            $.request.headers.vary,
            {'elements': ['Accept', 'Origin']}
          )
        ),
        :"$.x-ids" (
          %http:list (
            $.request.headers.x-ids,
            {'elements': ['1', '2']}
          )
        )
      )
    )
  )
)
