  :"get orders as a list" (

      :headers (
        %expect:object (
          $.request.headers => {'accept': ['application/json, text/plain'], 'x-ids': ['1,,2,'], 'vary': ['Accept', 'Origin']}
        ) => BOOL(true),
        :"$.accept" (
          %http:list (

          ) => BOOL(true)
        ) => BOOL(true),
        :"$.vary" (
          %http:list (
            $.request.headers.vary => ['Accept', 'Origin'],
            {'elements': ['Accept', 'Origin']}
          ) => BOOL(true)
        ) => BOOL(true),
        :"$.x-ids" (
          %http:list (
            $.request.headers.x-ids => ['1,,2,'],
            {'elements': ['1', '2']}
          ) => BOOL(true)
        ) => BOOL(true)
      ) => BOOL(true)
    ) => BOOL(true)
  ) => BOOL(true)
