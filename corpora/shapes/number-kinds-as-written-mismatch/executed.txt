        %expect:object (
          $.response.body => {'count': 100.0, 'scaled': 100.0, 'amount': 100}
        ) => BOOL(true),

            $.response.body.amount => 100

        ) => BOOL(false),
        :"$.count" (
          %match:integer (
            $.response.body.count => 100.0
          ) => ERROR(Expected an integer but got 100.0)
        ) => BOOL(false),
        :"$.scaled" (
          %match:integer (
            $.response.body.scaled => 100.0
          ) => ERROR(Expected an integer but got 100.0)
        ) => BOOL(false)
      ) => BOOL(false)
    ) => BOOL(false)
  ) => BOOL(false)
